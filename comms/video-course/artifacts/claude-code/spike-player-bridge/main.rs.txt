use std::fs::OpenOptions;
use std::io::Write;
use std::time::{Duration, Instant};
use tauri::{Manager, WebviewUrl, WebviewWindowBuilder};

fn log(line: &str) {
    if let Ok(mut f) = OpenOptions::new().create(true).append(true).open("spike.log") {
        let _ = writeln!(f, "{line}");
    }
}

#[tauri::command]
fn bridge_report(ev: String, t: f64, paused: bool, duration: f64, note: String) {
    // tick 太密，只记整秒变化之外的事件和每 5 个 tick 里的一条
    static N: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    if ev == "tick" && N.fetch_add(1, std::sync::atomic::Ordering::Relaxed) % 10 != 0 {
        return;
    }
    log(&format!("REPORT ev={ev} t={t:.2} paused={paused} dur={duration:.0} {note}"));
}

fn main() {
    let _ = std::fs::remove_file("spike.log");
    tauri::Builder::default()
        .invoke_handler(tauri::generate_handler![bridge_report])
        .setup(|app| {
            let url = "https://player.bilibili.com/player.html?bvid=BV1Ls411w7rx&p=2&autoplay=0";
            let w = WebviewWindowBuilder::new(app, "player", WebviewUrl::External(url.parse().unwrap()))
                .title("player")
                .inner_size(960.0, 600.0)
                .initialization_script(include_str!("bridge.js"))
                .build()?;
            let handle = app.handle().clone();
            std::thread::spawn(move || {
                let start = Instant::now();
                let step = |msg: &str, js: &str| {
                    log(&format!("[{:>5.1}s] {msg}", start.elapsed().as_secs_f32()));
                    let _ = w.eval(js);
                };
                std::thread::sleep(Duration::from_secs(10));
                step("A: seek 85 + play，观察 25 秒", "window.__athena.seek(85); window.__athena.play()");
                std::thread::sleep(Duration::from_secs(25));
                step("A2: 若被暂停则 play 续播，再观察 15 秒", "window.__athena.play().then(()=>window.__athena.ping())");
                std::thread::sleep(Duration::from_secs(15));
                step("B: seek 150 + 暂停点 155 + play", "window.__athena.seek(150); window.__athena.setPausePoint(155); window.__athena.play()");
                std::thread::sleep(Duration::from_secs(12));
                step("done", "window.__athena.ping()");
                std::thread::sleep(Duration::from_secs(2));
                handle.exit(0);
            });
            let _ = app.get_webview_window("main");
            Ok(())
        })
        .run(tauri::generate_context!())
        .expect("run");
}
